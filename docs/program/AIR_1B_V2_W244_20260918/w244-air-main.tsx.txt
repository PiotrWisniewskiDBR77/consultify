import '../src/index.css';
import React from 'react';
import {createRoot} from 'react-dom/client';
import {MemoryRouter} from 'react-router-dom';
import i18n from '../src/i18n';
import {InterviewHub} from '../src/components/Interview/InterviewHub';
import {FeatureFlagsProvider} from '../src/contexts/FeatureFlagsContext';
import {useAppStore} from '../src/store/useAppStore';
import {Api} from '../src/services/api';
import {V8InterviewApi} from '../src/services/api/v8/interview';
import records from '../docs/program/AIR_1B_V2_W244_20260918/real-records.json';
const p=new URLSearchParams(location.search), theme=p.get('theme')==='dark'?'dark':'light';
document.documentElement.classList.toggle('dark',theme==='dark');
document.documentElement.dataset.theme=theme;
useAppStore.setState({theme,currentUser:{id:records[0].owner_id,organizationId:records[0].organization_id,role:'SUPERADMIN',isAuthenticated:true} as any,currentOrganization:{id:records[0].organization_id,name:'Interview review proof'} as any});
const sessions=records.map(r=>({id:r.session_id,name:r.name,status:r.status,assignmentStatus:r.status,ownerId:r.owner_id,answeredQuestions:r.answered_questions,totalQuestions:r.total_questions,startedAt:r.started_at,lastActivityAt:r.last_activity_at}));
const assignments=records.map(r=>({id:r.id,sessionId:r.session_id,status:r.status,sessionName:r.name,assigneeUserId:r.assignee_user_id,aiReview:r.ai_review_snapshot_json,aiReviewedAt:r.ai_reviewed_at,session:sessions.find(s=>s.id===r.session_id)}));
Object.assign(V8InterviewApi,{getSessions:async()=>({sessions}),getManagedSessions:async()=>({sessions}),getManagedAssignments:async()=>({assignments}),getMyAssignments:async()=>({assignments:[]}),getOverdueAssignments:async()=>({assignments:[]}),listInsights:async()=>({insights:[]})});
Object.assign(Api,{getProjects:async()=>[],get:async(path:string)=>path==='/interview/sessions'?sessions:path.includes('memberships')?{memberships:[]}:path.includes('access/effective')?{effectiveAccess:{capabilities:['*']}}:[]});
await i18n.changeLanguage('en');
createRoot(document.getElementById('root')!).render(<FeatureFlagsProvider config={{enableLocalOverrides:true}} showDevTools={false}><MemoryRouter initialEntries={['/interview?tab='+(p.get('tab')||'managed')]}><div className="h-screen w-screen bg-c-surface"><InterviewHub/></div></MemoryRouter></FeatureFlagsProvider>);
